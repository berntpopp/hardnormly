**Step 2: Create the err_handler function.**

Place it early in the script (before any other functions). Pattern from research:

```bash
err_handler() {
    local exit_code=$?
    local line_number=$1
    local failed_command="${BASH_COMMAND}"
    # Write to stderr so it doesn't pollute stdout VCF output
    echo "ERROR: '${failed_command}' failed (exit ${exit_code}) at line ${line_number}" >&2
}
```

**Step 3: Create the cleanup_handler function.**

Replace the existing `cleanup_tmp_dir` function (lines 31-41) with a combined cleanup handler:

cleanup_handler() {

Replace the entire filter pipeline section (approximately lines 330-407 in the original script) with an eval-free approach using sequential bcftools calls and BCF temp files.

**Design (from research Pattern 5):**

The current code builds a `pipeline_cmd` string by concatenating bcftools commands with pipes, then runs `eval "$pipeline_cmd"`. Replace with:

1. **Initial step** — run `bcftools view | bcftools +fill-tags` to create the starting BCF:

```bash
bcftools view "$normalized_vcf" \
    | bcftools +fill-tags -Ob -o "$tmp_dir/filter_current.bcf" \
    || { log_msg "Error: Failed to initialize filter pipeline."; exit 1; }
```

2. **Build filter stages array** — encode each filter as `"name|action|expression"`:

```bash
filter_stages=()

# Region-based filters
if [[ -f "$tmp_dir/merged_include_regions.bed.gz" ]]; then
    filter_stages+=("NOT_IN_INCLUDE_REGION|e|INFO/INCLUDE_REGION!=1")
fi
if [[ -f "$tmp_dir/merged_exclude_regions.bed.gz" ]]; then
    filter_stages+=("IN_EXCLUDE_REGION|e|INFO/EXCLUDE_REGION=1")
fi

# Inline filters
for filter in "${filters[@]}"; do
    local filter_name filter_action filter_expr
    IFS=" " read -r filter_name filter_action filter_expr <<< "$filter"
    filter_stages+=("${filter_name}|${filter_action}|${filter_expr}")
done

# File-based filters
if [[ -n "$filters_file" ]]; then
    while IFS=" " read -r filter_name filter_action filter_expr; do
        filter_expr=$(tr -d '\r\n' <<< "$filter_expr")
        filter_stages+=("${filter_name}|${filter_action}|${filter_expr}")
    done < "$filters_file"
fi
```

3. **Apply filter stages sequentially** via BCF temp files:

```bash
for stage in "${filter_stages[@]}"; do
    local stage_name stage_action stage_expr
    IFS="|" read -r stage_name stage_action stage_expr <<< "$stage"
    debug_msg "Applying filter: $stage_name ($stage_action) '$stage_expr'"
    bcftools filter -m+ -s "$stage_name" "-${stage_action}" "$stage_expr" \
        "$tmp_dir/filter_current.bcf" -Ob -o "$tmp_dir/filter_next.bcf" \
        || { log_msg "Error: Filter '$stage_name' failed."; exit 1; }
    mv "$tmp_dir/filter_next.bcf" "$tmp_dir/filter_current.bcf"
done
```

Note: The `-${stage_action}` flag expands to `-e` (exclude) or `-i` (include), matching bcftools filter syntax. The expression goes as a direct argument — no quoting gymnastics needed because the shell handles it correctly when passed as `"$stage_expr"`.

4. **Final output** — determine output type and write:

```bash
# Build output arguments
local output_args=()
if [[ -n "$output_vcf" ]]; then
    if [[ "$output_vcf" == *.vcf.gz ]]; then
        output_args+=("-Oz")
        if $auto_index; then
            output_args+=("-W" "tbi")
            debug_msg "Auto-index enabled for compressed output."
        fi
    elif [[ "$output_vcf" == *.vcf ]]; then
        output_args+=("-Ov")
    else
        log_msg "Error: Unrecognized output file format for $output_vcf."
        exit 1
    fi
    output_args+=("-o" "$output_vcf")
else
    # stdout output (no -o flag)
    :
fi

# Apply PASS filter and write output
if $only_pass; then
    bcftools view -f PASS "${output_args[@]}" "$tmp_dir/filter_current.bcf" \
        || { log_msg "Error: PASS filter failed."; exit 1; }
else
    bcftools view "${output_args[@]}" "$tmp_dir/filter_current.bcf" \
        || { log_msg "Error: Failed to write output."; exit 1; }
fi
```
