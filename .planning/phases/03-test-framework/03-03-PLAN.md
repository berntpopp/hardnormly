**GROUP 1: Integration tests (TFWK-04)**

Test the full pipeline on real 1000G data. This exercises the complete pipeline including VCF normalization (bcftools norm with real multiallelic sites), BED processing, and filtering.

```bash
@test "TFWK-04: full pipeline on 1kg_NA12878 produces valid output" {
    local output="${TEST_TMP}/1kg_integration.vcf.gz"

    run "$HARDNORMLY" \
        -v "${REAL_DATA}/1kg_NA12878_chr22_16M.vcf.gz" \
        -f "${REAL_DATA}/chr22_16M_ref.fa" \
        -g "${TEST_DATA}/hg19_chr22.genome" \
        --filters-file "${REPO_ROOT}/defaults/gatk_filters.txt" \
        -o "$output" \
        --auto-index
    assert_success

    # Output exists and is valid VCF
    assert_file_exists "$output"
    run bcftools view -H "$output"
    assert_success

    # Output is non-empty (at least 1 variant)
    local count
    count=$(bcftools view -H "$output" | wc -l)
    [ "$count" -gt 0 ]
}
```

**Note:** This test runs WITHOUT BED files because the real data coordinates (16M+) don't match the synthetic BED regions (10k-90k). Running without BED files is a valid pipeline mode — it skips region annotation. If `hardnormly.sh` requires BED files, test with the GIAB high-confidence BED instead.

**GROUP 2: Regression tests (TFWK-05)**

Compare GATK pipeline output against expected golden file. Strip ##header lines, exact diff on data lines.

**PREREQUISITE:** Expected output files must exist from Phase 2 plan 02-04. Check for their existence:

```bash
@test "TFWK-05: GATK regression - output matches expected golden file" {
    # Skip if expected output doesn't exist yet (Phase 2 plan 04 not complete)
    if [[ ! -f "${EXPECTED}/gatk_sample_A_filtered.vcf.gz" ]]; then
        skip "Expected output not yet generated (run Phase 2 plan 04)"
    fi

    local output="${TEST_TMP}/gatk_regression.vcf.gz"

    run "$HARDNORMLY" \
        -v "${SYNTH}/gatk_sample_A.vcf.gz" \
        -f "${SYNTH}/mini_ref.fa" \
        -b "${TEST_DATA}/include_regions.bed" \
        -e "${TEST_DATA}/exclude_regions.bed" \
        --filters-file "${REPO_ROOT}/defaults/gatk_filters.txt" \
        -g "${TEST_DATA}/hg19_chr22.genome" \
        -o "$output" \
        --auto-index
    assert_success

    # Strip headers, compare data lines
    bcftools view -H "$output" > "${TEST_TMP}/actual_data.vcf"
    bcftools view -H "${EXPECTED}/gatk_sample_A_filtered.vcf.gz" > "${TEST_TMP}/expected_data.vcf"

    if ! diff -u "${TEST_TMP}/expected_data.vcf" "${TEST_TMP}/actual_data.vcf" > "${TEST_TMP}/diff.txt" 2>&1; then
        local changed_lines
        changed_lines=$(grep -c '^[+-]' "${TEST_TMP}/diff.txt" || true)
        echo "Regression failure: ${changed_lines} changed lines"
        echo "=== First 20 differences ==="
        head -20 "${TEST_TMP}/diff.txt"
        fail "Output does not match expected (see diff above)"
    fi
}
```
