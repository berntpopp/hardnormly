2. **normalize_vcf()** — extracted from hardnormly.sh Step 5 (lines 384-412):

   - This function does NOT use `run_cmd` (Research Pitfall 6: bcftools norm emits warnings on stderr even on success)

     a. Create temp files for stdout and stderr capture (in `$tmp_dir`)
     b. Run `bcftools norm -m-any --force -a --atom-overlaps . --write-index=tbi -f "$fasta_file" "$vcf_file" -Oz -o "$output_vcf"` with stderr and stdout captured
     c. On failure: log error + stderr details, clean up temp files, return 1
     d. On success: check for "Warning" in stderr and log if present, check for "Lines" in stdout and log summary, clean up temp files
   - Important: temp files for norm output must be in `$tmp_dir` (not bare `mktemp`) so cleanup_handler covers them per REFR-12. Use `mktemp "${tmp_dir}/norm-stderr-XXXXXX"` and `mktemp "${tmp_dir}/norm-stdout-XXXXXX"`.
   - Remove the inline `rm -f "$norm_output" "$norm_stdout"` at error sites — cleanup_handler handles them since they're in $tmp_dir now (REFR-12).

   ```bash
   normalize_vcf() {
       local vcf_file="$1"
       local fasta_file="$2"
       local output_vcf="$3"
       local tmp_dir="$4"

       local norm_stderr norm_stdout
       norm_stderr=$(mktemp "${tmp_dir}/norm-stderr-XXXXXX")
       norm_stdout=$(mktemp "${tmp_dir}/norm-stdout-XXXXXX")

       bcftools norm -m-any --force -a --atom-overlaps . --write-index=tbi \
           -f "$fasta_file" "$vcf_file" -Oz -o "$output_vcf" \
           2>"$norm_stderr" 1>"$norm_stdout" \
           || {
               error_msg "Failed to normalize the VCF"
               error_msg "bcftools norm details: $(< "$norm_stderr")"
               return 1
           }

       if grep -q "Warning" "$norm_stderr"; then
           log_msg "bcftools norm warnings: $(< "$norm_stderr")"
       fi
       if grep -q "Lines" "$norm_stdout"; then
           log_msg "bcftools norm summary: $(grep 'Lines' "$norm_stdout")"
       fi

       rm -f "$norm_stderr" "$norm_stdout"
       debug_msg "Normalized VCF written to: ${output_vcf}"
   }
   ```

3. **apply_filter_stages()** — extract from lines 453-467:
   - Args: `tmp_dir`, then the filter_stages array entries (passed as remaining args)
   - Iterates over stages, applies each with `run_cmd bcftools filter ...`
   - Each stage: parse `name|action|expression`, run bcftools filter from filter_current.bcf to filter_next.bcf, mv next to current
   ```bash
   apply_filter_stages() {
       local tmp_dir="$1"
       shift
       local -a stages=("$@")

       for stage in "${stages[@]}"; do
           local stage_name stage_action stage_expr
           IFS="|" read -r stage_name stage_action stage_expr <<< "$stage"
           debug_msg "Applying filter: ${stage_name} (${stage_action}) '${stage_expr}'"
           run_cmd bcftools filter -m+ -s "$stage_name" "-${stage_action}" "$stage_expr" \
               "${tmp_dir}/filter_current.bcf" -Ob -o "${tmp_dir}/filter_next.bcf" \
               || { error_msg "Filter '${stage_name}' failed"; return 1; }
           mv "${tmp_dir}/filter_next.bcf" "${tmp_dir}/filter_current.bcf"
       done
   }
   ```

4. **write_filtered_output()** — extract from lines 469-500 (output format detection, PASS filter, final write):
   - Args: `tmp_dir`, `output_vcf`, `only_pass` (true/false string), `auto_index` (true/false string)
   - Builds output_args array based on output_vcf extension
   - Applies PASS filter if only_pass is true
   - Writes final output via `run_cmd bcftools view ...`
   ```bash
   write_filtered_output() {
       local tmp_dir="$1"
       local output_vcf="$2"
       local only_pass="$3"
       local auto_index="$4"

       local -a output_args=()
       if [[ -n "$output_vcf" ]]; then
           if [[ "$output_vcf" == *.vcf.gz ]]; then
               output_args+=("-Oz")
               if [[ "$auto_index" == "true" ]]; then
                   output_args+=("--write-index=tbi")
                   debug_msg "Auto-index enabled for compressed output."
               fi
           elif [[ "$output_vcf" == *.vcf ]]; then
               output_args+=("-Ov")
           else
               error_msg "Unrecognized output file format for ${output_vcf}"
               return 1
           fi
           output_args+=("-o" "$output_vcf")
       fi

       if [[ "$only_pass" == "true" ]]; then
           run_cmd bcftools view -f PASS "${output_args[@]}" "${tmp_dir}/filter_current.bcf" \
               || { error_msg "PASS filter failed"; return 1; }
       else
           run_cmd bcftools view "${output_args[@]}" "${tmp_dir}/filter_current.bcf" \
               || { error_msg "Failed to write output"; return 1; }
       fi
   }
   ```
